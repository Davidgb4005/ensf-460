name Alarm and acknowledge

allow state 500ms
allow previous_state 500ms
allow elapsed 1200ms
allow pb1 250ms
allow pb2 250ms
allow led1_state 500ms
allow led1_output 1200ms
allow led2_state 500ms
allow led2_output 500ms

# Reset, create a short preset, let it expire, then acknowledge the alarm.
hold pb3 4s
hold pb1+pb2 4s
wait 300ms
hold pb1 400ms
wait 300ms
hold pb1+pb2 200ms
wait 4s
hold pb1 200ms
wait 500ms
