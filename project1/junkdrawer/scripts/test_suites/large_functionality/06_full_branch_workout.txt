# Full branch workout.
#
# This is intentionally a long-running functional test. With the default
# startup reset it runs for approximately five minutes. It exercises every
# digital-twin main-loop branch, including short and long increments,
# decrement combinations, bounds, invalid starts, reset, run, pause/resume,
# pause-clear, alarm acknowledgement, and ignored alarm inputs.

note ===== INITIAL STATE AND INVALID STARTS =====
expect state IDLE
expect preset 0
expect elapsed 0
expect led1_state OFF
expect led2_state OFF

# PB1+PB2 cannot start with a zero preset.
press pb1+pb2 0.30
wait 0.90
expect state IDLE
expect preset 0
expect elapsed 0

# PB2+PB3 at zero must remain clamped at zero.
press pb2+pb3 0.50
wait 1.00
expect state IDLE
expect preset 0
expect_range preset 0 0

# All three buttons have no dedicated IDLE action and must not start the timer.
press pb1+pb2+pb3 0.50
wait 1.00
expect state IDLE
expect preset 0

# A zero-preset start attempt leaves the start trigger armed in the current
# implementation. Hold PB1+PB2 through reset_delay before testing increments.
press pb1+pb2 3.50
wait 4.20
expect state IDLE
expect preset 0
expect elapsed 0

note ===== SHORT AND LONG SECOND INCREMENTS =====
# Short PB1 increments use one-second steps.
press pb1 1.00
wait 1.50
expect state IDLE
expect_range preset 3 8

# Holding PB1 crosses fast_increment_count and selects long-second steps.
press pb1 4.80
wait 5.30
expect state IDLE
expect_range preset 20 90

note ===== SHORT AND LONG MINUTE INCREMENTS =====
# PB2 selects minute increments. The exact value is timing-sensitive, so the
# test checks that the value increased substantially and remains in range.
press pb2 1.00
wait 1.50
expect state IDLE
expect_range preset 240 420

# Holding PB2 crosses the fast-increment threshold and uses long-minute steps.
press pb2 4.80
wait 5.30
expect state IDLE
expect_range preset 1200 3599

note ===== SECOND AND MINUTE DECREMENT BRANCHES =====
# PB1+PB3 selects short/long second decrement.
press pb1+pb3 1.00
wait 1.50
expect state IDLE
expect_range preset 1150 3599

press pb1+pb3 4.80
wait 5.30
expect state IDLE
expect_range preset 1050 3599

# PB2+PB3 selects short/long minute decrement.
press pb2+pb3 1.00
wait 1.50
expect state IDLE
expect_range preset 900 3599

press pb2+pb3 4.80
wait 5.30
expect state IDLE
expect_range preset 0 3599

note ===== MAXIMUM AND MINIMUM CLAMPING =====
# A long PB2 hold must clamp at the configured maximum of 3599 seconds.
press pb2 10.50
wait 11.00
expect state IDLE
expect preset 3599
expect_range preset 3599 3599

# A long PB2+PB3 hold must clamp at zero.
press pb2+pb3 10.50
wait 11.00
expect state IDLE
expect preset 0
expect_range preset 0 0

note ===== LONG RESET FROM IDLE =====
# Build a nonzero preset, then hold PB1+PB2 long enough to clear it.
press pb1 1.00
wait 1.50
expect_range preset 3 8
press pb1+pb2 3.50
wait 4.20
expect state IDLE
expect preset 0
expect elapsed 0

note ===== LONG INTERVAL RUN AND PAUSE =====
# Create a roughly one-minute interval using minute increments.
press pb2 0.80
wait 1.40
expect_range preset 120 300
press pb1+pb2 0.30
wait_until state RUN 2.00
wait 0.10
expect led1_state BLINK
expect led2_state OFF

# Let the timer count for a while and verify elapsed is decreasing.
wait 8.00
expect state RUN
expect_range elapsed 100 300

# Continue a sustained run long enough to exercise repeated one-second
# countdown updates and LED blinking over a longer interval.
wait 35.00
expect state RUN
expect_range elapsed 60 300

# Pause and verify elapsed remains stable during a long pause.
press pb3 0.25
wait_until state PAUSE 2.00
expect state PAUSE
expect led1_state OFF
wait 20.00
expect state PAUSE

# A second PB3 short press resumes from pause.
press pb3 0.25
wait_until state RUN 2.00
expect state RUN
wait 30.00
expect state RUN

# Pause again and clear with a long PB3 hold.
press pb3 0.25
wait_until state PAUSE 2.00
expect state PAUSE
wait 0.40
press pb3 3.50
wait_until state IDLE 5.00
wait 0.75
expect state IDLE
expect elapsed 0
expect_range preset 120 300

note ===== SECOND SUSTAINED INTERVAL =====
# Repeat a long run after the first clear to catch stale timing, LED, or
# transition state left behind by the previous cycle.
press pb1+pb2 0.30
wait_until state RUN 2.00
wait 35.00
expect state RUN
expect_range elapsed 40 300
press pb3 0.25
wait_until state PAUSE 2.00
wait 15.00
expect state PAUSE
press pb3 0.25
wait_until state RUN 2.00
wait 25.00
expect state RUN
press pb3 3.50
wait_until state IDLE 5.00
wait 0.75
expect state IDLE
expect elapsed 0
expect_range preset 120 300

note ===== PAUSE CLEAR WITH PB3 HELD THROUGH THE DELAY =====
# Start again, pause, and keep PB3 held beyond clear_delay.
press pb1+pb2 0.30
wait_until state RUN 2.00
press pb3 3.80
wait_until state IDLE 5.00
wait 1.00
expect state IDLE
expect elapsed 0
expect_range preset 120 300

note ===== RESET AFTER A PAUSE CLEAR =====
press pb1+pb2 3.50
wait 4.20
expect state IDLE
expect preset 0
expect elapsed 0

note ===== SHORT RUN TO ALARM AND PB3-ONLY ALARM INPUT =====
# Make a small preset and run it to completion.
press pb1 0.65
wait 1.10
expect_range preset 1 5
press pb1+pb2 0.30
wait_until state RUN 2.00
wait 0.10
expect led1_state BLINK
wait_until state ALARM 8.00
wait 0.10
expect state ALARM
expect elapsed 0
expect led1_state ON
expect led2_state BLINK

# PB3 alone must not acknowledge an alarm.
press pb3 0.30
wait 1.00
expect state ALARM
expect led1_state ON
expect led2_state BLINK

# PB1 acknowledges the alarm and returns to idle.
press pb1 0.10
wait_until state IDLE 2.00
wait 0.20
expect state IDLE
expect led1_state OFF
expect led2_state OFF

note ===== ALARM ACKNOWLEDGEMENT THROUGH PB2 =====
# Restart from the retained preset and acknowledge with PB2 this time.
press pb1+pb2 0.30
wait_until state RUN 2.00
wait_until state ALARM 8.00
wait 0.10
expect state ALARM
press pb2 0.10
wait_until state IDLE 2.00
wait 0.20
expect state IDLE
expect elapsed 0

note ===== REPEATED PAUSE AND RESUME CYCLES =====
# Increase the preset slightly, then repeat pause/resume several times.
press pb1 0.80
wait 1.30
expect_range preset 3 10
press pb1+pb2 0.30
wait_until state RUN 2.00

press pb3 0.20
wait_until state PAUSE 2.00
wait 2.00
expect state PAUSE
press pb3 0.20
wait_until state RUN 2.00
wait 0.35

press pb3 0.20
wait_until state PAUSE 2.00
wait 1.50
expect state PAUSE
press pb3 0.20
wait_until state RUN 2.00
wait 0.35

press pb3 0.20
wait_until state PAUSE 2.00
wait 1.50
expect state PAUSE
press pb3 0.20
wait_until state RUN 2.00
wait_until state ALARM 12.00
expect state ALARM

note ===== ALARM ACKNOWLEDGEMENT AFTER REPEATED PAUSES =====
press pb1 0.10
wait_until state IDLE 2.00
wait 0.20
expect state IDLE
expect elapsed 0

note ===== FINAL RESET AND POST-RESET INVALID ACTIONS =====
press pb1+pb2 3.50
wait 4.20
expect state IDLE
expect preset 0
expect elapsed 0

# Confirm decrement and start remain harmless after the final reset.
press pb1+pb3 1.00
wait 1.50
press pb1+pb2 0.30
wait 1.00
expect state IDLE
expect preset 0
expect elapsed 0

note ===== COMPLETE =====
