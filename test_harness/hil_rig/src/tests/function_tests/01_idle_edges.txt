functional_idle_edges
# Boot must settle in idle.
reset
wait 1500
assert idle

# Starting with a zero preset must remain in idle.
press 150 1 2
wait 500
assert idle

# Decrementing below zero must clamp the preset and remain in idle.
press 900 1 3
wait 300
assert idle
press 900 2 3
wait 300
assert idle

# A short PB1+PB2 press with a non-zero preset starts a run.
press 450 1
wait 200
assert idle
press 150 1 2
wait 500
assert run

# A fresh reset must return to idle from an active run.
reset
wait 1500
assert idle
